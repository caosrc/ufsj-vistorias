# 9. src/components/Mapa.jsx

```jsx
import { useEffect, useRef } from 'react'
import maplibregl from 'maplibre-gl'
import 'maplibre-gl/dist/maplibre-gl.css'

export default function Mapa() {
  const mapContainer = useRef(null)
  const map = useRef(null)

  useEffect(() => {
    if (map.current) return

    map.current = new maplibregl.Map({
      container: mapContainer.current,
      style: {
        version: 8,
        sources: {
          osm: {
            type: 'raster',
            tiles: [
              'https://tile.openstreetmap.org/{z}/{x}/{y}.png'
            ],
            tileSize: 256
          }
        },
        layers: [
          {
            id: 'osm',
            type: 'raster',
            source: 'osm'
          }
        ]
      },
      center: [-43.6919, -20.5236],
      zoom: 13
    })

    map.current.addControl(new maplibregl.NavigationControl())

    map.current.on('click', (e) => {
      const lat = e.lngLat.lat
      const lng = e.lngLat.lng

      localStorage.setItem(
        'ultimaCoordenada',
        JSON.stringify({ lat, lng })
      )

      new maplibregl.Marker({ color: 'red' })
        .setLngLat([lng, lat])
        .addTo(map.current)

      alert(`Coordenada salva:\nLat: ${lat}\nLng: ${lng}`)
    })

  }, [])

  return (
    <div
      ref={mapContainer}
      style={{ width: '100%', height: '100vh' }}
    />
  )
}
```

# 10. ADICIONAR CAMADA SATÉLITE

Substituir em Mapa.jsx:

```jsx
sources: {
  osm: {
    type: 'raster',
    tiles: ['https://tile.openstreetmap.org/{z}/{x}/{y}.png'],
    tileSize: 256
  },

  satelite: {
    type: 'raster',
    tiles: [
      'https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile/{z}/{y}/{x}'
    ],
    tileSize: 256
  }
}
```

# 13. CADASTRO DE OCORRÊNCIAS

# src/pages/Ocorrencias.jsx

```jsx
import { useState } from 'react'

export default function Ocorrencias() {
  const [tipo, setTipo] = useState('')
  const [descricao, setDescricao] = useState('')

  function salvar() {
    const ocorrencias = JSON.parse(localStorage.getItem('ocorrencias')) || []

    const coordenada = JSON.parse(localStorage.getItem('ultimaCoordenada'))

    ocorrencias.push({
      tipo,
      descricao,
      coordenada,
      data: new Date()
    })

    localStorage.setItem('ocorrencias', JSON.stringify(ocorrencias))

    alert('Ocorrência salva!')
  }

  return (
    <div style={{ padding: 20 }}>
      <h1>Ocorrências</h1>

      <input
        placeholder='Tipo'
        value={tipo}
        onChange={(e) => setTipo(e.target.value)}
      />

      <br /><br />

      <textarea
        placeholder='Descrição'
        value={descricao}
        onChange={(e) => setDescricao(e.target.value)}
      />

      <br /><br />

      <button onClick={salvar}>
        Salvar
      </button>
    </div>
  )
}
```

# 14. MENU DE VISTORIA

# src/pages/Vistoria.jsx

```jsx
import { useState } from 'react'

export default function Vistoria() {
  const [nome, setNome] = useState('')
  const [solo, setSolo] = useState('')
  const [risco, setRisco] = useState('')
  const [observacao, setObservacao] = useState('')

  function salvar() {
    const dados = JSON.parse(localStorage.getItem('vistorias')) || []

    const coordenada = JSON.parse(localStorage.getItem('ultimaCoordenada'))

    dados.push({
      nome,
      solo,
      risco,
      observacao,
      coordenada
    })

    localStorage.setItem('vistorias', JSON.stringify(dados))

    alert('Vistoria salva!')
  }

  return (
    <div style={{ padding: 20 }}>
      <h1>Vistoria Residencial</h1>

      <input
        placeholder='Responsável'
        value={nome}
        onChange={(e) => setNome(e.target.value)}
      />

      <br /><br />

      <select
        value={solo}
        onChange={(e) => setSolo(e.target.value)}
      >
        <option>Tipo de solo</option>
        <option>Argiloso</option>
        <option>Arenoso</option>
        <option>Rocha</option>
        <option>Aterro</option>
      </select>

      <br /><br />

      <select
        value={risco}
        onChange={(e) => setRisco(e.target.value)}
      >
        <option>Grau de risco</option>
        <option>R1</option>
        <option>R2</option>
        <option>R3</option>
        <option>R4</option>
      </select>

      <br /><br />

      <textarea
        placeholder='Observações'
        value={observacao}
        onChange={(e) => setObservacao(e.target.value)}
      />

      <br /><br />

      <button onClick={salvar}>
        Salvar Vistoria
      </button>
    </div>
  )
}
```
