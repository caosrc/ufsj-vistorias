export class RainEngine {

  constructor(map) {
    this.map = map;
    this.gotas = [];
  }

  iniciar(intensidade = 500) {

    const bounds = this.map.getBounds();

    this.gotas = [];

    for (let i = 0; i < intensidade; i++) {

      this.gotas.push({
        lat: bounds.getNorth() - Math.random() * (bounds.getNorth() - bounds.getSouth()),
        lng: bounds.getWest() + Math.random() * (bounds.getEast() - bounds.getWest()),
        v: 0.0005 + Math.random() * 0.001
      });
    }
  }

  update() {

    const bounds = this.map.getBounds();

    this.gotas.forEach(g => {

      g.lat -= g.v;

      if (g.lat < bounds.getSouth()) {
        g.lat = bounds.getNorth();
      }
    });
  }

  render(layer) {

    layer.clearLayers();

    this.gotas.forEach(g => {

      L.polyline([
        [g.lat, g.lng],
        [g.lat - 0.001, g.lng]
      ], {
        color: "rgba(0,150,255,0.3)",
        weight: 1
      }).addTo(layer);

    });
  }
}
🌊 2. FLUXO CONTÍNUO (ÁGUA ANDANDO)
💻 arquivo: flowEngine.js
export class FlowEngine {

  constructor(grid) {
    this.grid = grid;
    this.particulas = [];
  }

  iniciar() {

    this.particulas = [];

    this.grid.forEach(c => {

      if (c.acum > 50) {

        this.particulas.push({
          lat: c.lat,
          lng: c.lng,
          vel: 0.3,
          vida: 200
        });

      }

    });
  }

  step(encontrarCelula) {

    this.particulas.forEach(p => {

      const c = encontrarCelula(p.lat, p.lng);

      if (!c || !c.fluxo) return;

      const dx = c.fluxo.lng - c.lng;
      const dy = c.fluxo.lat - c.lat;

      const slope = Math.max(c.elev - c.fluxo.elev, 0.001);

      const v = Math.sqrt(slope) * p.vel;

      p.lat += dy * v;
      p.lng += dx * v;

      p.vida--;

    });

    this.particulas = this.particulas.filter(p => p.vida > 0);
  }

  render(layer) {

    layer.clearLayers();

    this.particulas.forEach(p => {

      L.circleMarker([p.lat, p.lng], {
        radius: 2,
        color: "cyan",
        fillOpacity: 0.7
      }).addTo(layer);

    });
  }
}

export function calcularAcumulo(grid) {

  grid.forEach(c => {

    if (c.acum > 100) {
      c.agua += c.acum * 0.1;
    }

  });
}
🎨 renderização contínua (sem bolinhas)
export function renderFlood(grid, layer) {

  layer.clearLayers();

  grid.forEach(c => {

    if (c.agua > 20) {

      const cor =
        c.agua > 200 ? "red" :
        c.agua > 100 ? "purple" :
        c.agua > 50 ? "blue" :
        "lightblue";

      L.circle([c.lat, c.lng], {
        radius: 15,
        color: cor,
        fillOpacity: 0.4
      }).addTo(layer);
    }

  });
}

export function corVelocidade(c) {

  const slope = Math.max(c.elev - c.fluxo.elev, 0);

  if (slope > 20) return "red";
  if (slope > 10) return "orange";
  if (slope > 5) return "yellow";

  return "blue";
}

export class Timeline {

  constructor() {
    this.frames = [];
  }

  salvar(grid) {
    this.frames.push(JSON.parse(JSON.stringify(grid)));
  }

  reproduzir(callback) {

    let i = 0;

    const loop = () => {

      if (i >= this.frames.length) return;

      callback(this.frames[i]);

      i++;

      setTimeout(loop, 200);
    };

    loop();
  }
}

const rain = new RainEngine(map);
const flow = new FlowEngine(grid);
const timeline = new Timeline();

const rainLayer = L.layerGroup().addTo(map);
const flowLayer = L.layerGroup().addTo(map);
const floodLayer = L.layerGroup().addTo(map);

rain.iniciar(800);
flow.iniciar();

function loop() {

  rain.update();
  rain.render(rainLayer);

  flow.step(encontrarCelula);
  flow.render(flowLayer);

  calcularAcumulo(grid);
  renderFlood(grid, floodLayer);

  timeline.salvar(grid);

  requestAnimationFrame(loop);
}

loop();

function atualizar3D(mesh, grid) {

  const pos = mesh.geometry.attributes.position;

  for (let i = 0; i < pos.count; i++) {

    const c = grid[i];

    const agua = (c.agua || 0) * 0.02;

    pos.setZ(i, c.elev/50 + agua);
  }

  pos.needsUpdate = true;
}
