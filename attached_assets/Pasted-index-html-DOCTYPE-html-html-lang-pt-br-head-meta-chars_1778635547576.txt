// ===============================
// MAPA
// ===============================

const map = L.map('map').setView([-20.66, -43.79], 13);

// OpenStreetMap
L.tileLayer(
  'https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png',
  {
    attribution: '&copy; OpenStreetMap'
  }
).addTo(map);

// ===============================
// CURVAS DE NÍVEL
// ===============================

let contoursData;

fetch('contours.geojson')
  .then(res => res.json())
  .then(data => {

    contoursData = data;

    L.geoJSON(data, {

      style: function(feature) {

        return {
          color: '#8B4513',
          weight: 1
        };
      }

    }).addTo(map);

  });

// ===============================
// DRAW
// ===============================

const drawnItems = new L.FeatureGroup();

map.addLayer(drawnItems);

const drawControl = new L.Control.Draw({

  draw: {
    polygon: false,
    rectangle: false,
    circle: false,
    marker: false,
    circlemarker: false,
    polyline: true
  },

  edit: {
    featureGroup: drawnItems
  }

});

map.addControl(drawControl);

// ===============================
// PERFIL CHART
// ===============================

let chart;

// ===============================
// EVENTO DESENHO
// ===============================

map.on(L.Draw.Event.CREATED, async function(e) {

  drawnItems.clearLayers();

  const layer = e.layer;

  drawnItems.addLayer(layer);

  await analisarLinha(layer);

});

// ===============================
// ANÁLISE
// ===============================

async function analisarLinha(layer) {

  const latlngs = layer.getLatLngs();

  const coords = latlngs.map(p => [p.lng, p.lat]);

  const line = turf.lineString(coords);

  // distância total
  const totalDist =
    turf.length(line, { units: 'kilometers' }) * 1000;

  // ===============================
  // INTERSEÇÃO CURVAS
  // ===============================

  let elevations = [];

  contoursData.features.forEach(feature => {

    const inter = turf.lineIntersect(line, feature);

    if (inter.features.length > 0) {

      elevations.push(feature.properties.elevation);

    }

  });

  elevations = [...new Set(elevations)];

  elevations.sort((a,b)=>a-b);

  if (elevations.length < 2) {

    document.getElementById('info').innerHTML =
      'Linha cruzou poucas curvas';

    return;
  }

  // ===============================
  // DESNÍVEL
  // ===============================

  const minAlt = elevations[0];
  const maxAlt = elevations[elevations.length - 1];

  const desnivel = maxAlt - minAlt;

  const slope = (desnivel / totalDist) * 100;

  // ===============================
  // PERFIL ALTIMÉTRICO
  // ===============================

  let profileDistances = [];
  let profileElevations = [];

  const samples = 50;

  for (let i = 0; i <= samples; i++) {

    const segment = turf.along(
      line,
      (i / samples) * (totalDist / 1000),
      { units: 'kilometers' }
    );

    const point = segment.geometry.coordinates;

    // altitude simulada
    // depois você pode trocar por GeoTIFF real

    const elev =
      minAlt +
      ((maxAlt - minAlt) * (i / samples));

    profileDistances.push(
      ((i / samples) * totalDist).toFixed(0)
    );

    profileElevations.push(elev);
  }

  // ===============================
  // MOSTRAR INFO
  // ===============================

  document.getElementById('info').innerHTML = `

    <b>Distância:</b> ${totalDist.toFixed(1)} m<br>

    <b>Altitude mínima:</b> ${minAlt} m<br>

    <b>Altitude máxima:</b> ${maxAlt} m<br>

    <b>Desnível:</b> ${desnivel} m<br>

    <b>Declividade:</b> ${slope.toFixed(2)} %<br>

  `;

  desenharGrafico(
    profileDistances,
    profileElevations
  );

}

// ===============================
// GRÁFICO
// ===============================

function desenharGrafico(distances, elevations) {

  const ctx =
    document.getElementById('profileChart');

  if (chart) {
    chart.destroy();
  }

  chart = new Chart(ctx, {

    type: 'line',

    data: {

      labels: distances,

      datasets: [{

        label: 'Perfil Altimétrico',

        data: elevations,

        borderColor: 'red',

        backgroundColor:
          'rgba(255,0,0,0.2)',

        tension: 0.3,

        fill: true

      }]

    },

    options: {

      responsive: true,

      plugins: {

        legend: {
          display: true
        }

      },

      scales: {

        x: {

          title: {
            display: true,
            text: 'Distância (m)'
          }

        },

        y: {

          title: {
            display: true,
            text: 'Altitude (m)'
          }

        }

      }

    }

  });

}

  this._div.style.borderRadius = '6px';

};
