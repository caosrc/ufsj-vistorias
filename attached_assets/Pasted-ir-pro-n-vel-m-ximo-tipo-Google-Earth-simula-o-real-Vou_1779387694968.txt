export function criarMapa() {

  const map = new mapboxgl.Map({
    container: "map",
    style: "mapbox://styles/mapbox/satellite-streets-v12",
    center: [-43.69, -20.51], // Ouro Branco
    zoom: 13,
    pitch: 60,
    bearing: -20,
    antialias: true
  });

  map.on("load", () => {

    map.addSource("mapbox-dem", {
      type: "raster-dem",
      url: "mapbox://mapbox.terrain-rgb",
      tileSize: 512,
      maxzoom: 14
    });

    map.setTerrain({
      source: "mapbox-dem",
      exaggeration: 1.5
    });

  });

  return map;
}

import * as THREE from "three";

export function criarCamadaAgua(map) {

  const customLayer = {

    id: "water-layer",
    type: "custom",
    renderingMode: "3d",

    onAdd: function (map, gl) {

      this.camera = new THREE.Camera();
      this.scene = new THREE.Scene();

      const geometry = new THREE.PlaneGeometry(1000, 1000, 100, 100);

      const material = new THREE.ShaderMaterial({

        uniforms: {
          time: { value: 0 }
        },

        vertexShader: `
          varying vec2 vUv;
          void main() {
            vUv = uv;
            gl_Position = projectionMatrix * modelViewMatrix * vec4(position,1.0);
          }
        `,

        fragmentShader: `
          uniform float time;
          varying vec2 vUv;

          void main() {

            float wave = sin(vUv.x * 20.0 + time) * 0.1;

            gl_FragColor = vec4(0.0, 0.5, 1.0, 0.6 + wave);

          }
        `,

        transparent: true
      });

      this.mesh = new THREE.Mesh(geometry, material);
      this.scene.add(this.mesh);

      this.renderer = new THREE.WebGLRenderer({
        canvas: map.getCanvas(),
        context: gl
      });

      this.renderer.autoClear = false;
    },

    render: function (gl, matrix) {

      this.mesh.material.uniforms.time.value += 0.05;

      this.renderer.render(this.scene, this.camera);
      map.triggerRepaint();
    }
  };

  return customLayer;
}

export async function carregarRuas() {

  const query = `
  [out:json];
  way["highway"](around:8000,-20.51,-43.69);
  out geom;
  `;

  const res = await fetch("https://overpass-api.de/api/interpreter", {
    method: "POST",
    body: query
  });

  return (await res.json()).elements;
}

export function gerarFluxoRuas(ruas) {

  return ruas.map(r => {

    return r.geometry.map(p => [p.lon, p.lat]);

  });
}

import { criarMapa } from "./map.js";
import { criarCamadaAgua } from "./waterLayer.js";
import { carregarRuas, gerarFluxoRuas } from "./osmFlow.js";

const map = criarMapa();

map.on("load", async () => {

  // camada 3D água
  map.addLayer(criarCamadaAgua(map));

  // ruas
  const ruas = await carregarRuas();
  const fluxo = gerarFluxoRuas(ruas);

  // render fluxo
  map.addSource("fluxo", {
    type: "geojson",
    data: {
      type: "FeatureCollection",
      features: fluxo.map(f => ({
        type: "Feature",
        geometry: {
          type: "LineString",
          coordinates: f
        }
      }))
    }
  });

  map.addLayer({
    id: "fluxo",
    type: "line",
    source: "fluxo",
    paint: {
      "line-color": "#00ffff",
      "line-width": 2
    }
  });

});
