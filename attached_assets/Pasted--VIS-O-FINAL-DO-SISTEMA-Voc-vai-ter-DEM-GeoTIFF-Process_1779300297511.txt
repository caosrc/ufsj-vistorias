🧠 3. Código PROFISSIONAL (com DEM real)
🔥 Lê GeoTIFF + calcula slope
import GeoTIFF from 'geotiff';
import * as turf from '@turf/turf';
import fs from 'fs';

// =============================
// CONFIG
// =============================
const filePath = './dem.tif';
const cellSize = 5; // 5 metros (RECOMENDADO)

// =============================
// CARREGAR DEM
// =============================
const tiff = await GeoTIFF.fromFile(filePath);
const image = await tiff.getImage();

const width = image.getWidth();
const height = image.getHeight();

const raster = await image.readRasters();
const data = raster[0];

// bounding box
const [minX, minY, maxX, maxY] = image.getBoundingBox();

// resolução
const resX = (maxX - minX) / width;
const resY = (maxY - minY) / height;

// =============================
// FUNÇÃO PEGAR ELEVAÇÃO
// =============================
function getElevation(x, y) {
  const col = Math.floor((x - minX) / resX);
  const row = Math.floor((maxY - y) / resY);

  const index = row * width + col;
  return data[index];
}

// =============================
// CALCULAR SLOPE
// =============================
function calcSlope(z, zx, zy, d) {
  const dzdx = (zx - z) / d;
  const dzdy = (zy - z) / d;

  const slopeRad = Math.atan(Math.sqrt(dzdx**2 + dzdy**2));

  return {
    deg: slopeRad * (180 / Math.PI),
    perc: Math.tan(slopeRad) * 100
  };
}

// =============================
// GERAR GRID
// =============================
const bbox = [minX, minY, maxX, maxY];

const grid = turf.squareGrid(bbox, cellSize, { units: "meters" });

const resultado = [];

for (const cell of grid.features) {
  const center = turf.center(cell).geometry.coordinates;

  const [x, y] = center;

  const z = getElevation(x, y);
  const zx = getElevation(x + cellSize, y);
  const zy = getElevation(x, y + cellSize);

  if (z === undefined || zx === undefined || zy === undefined) continue;

  const slope = calcSlope(z, zx, zy, cellSize);

  resultado.push({
    ...cell,
    properties: {
      slope_deg: slope.deg,
      slope_perc: slope.perc
    }
  });
}

// =============================
// SALVAR
// =============================
fs.writeFileSync('slope.geojson', JSON.stringify({
  type: "FeatureCollection",
  features: resultado
}));

console.log("✅ slope.geojson gerado!");

🎯 Função de cor (engenharia)
function getColor(s) {
  if (s < 3) return "#00ff00";   // plano
  if (s < 8) return "#aaff00";
  if (s < 20) return "#ffff00";
  if (s < 45) return "#ff8800";
  return "#ff0000";             // muito íngreme
}
