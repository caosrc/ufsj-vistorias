🔹 Função para buscar chuva
async function getPrevisao(lat, lon) {
  const url = `https://api.open-meteo.com/v1/forecast?latitude=${lat}&longitude=${lon}&daily=precipitation_sum&timezone=auto`;

  const res = await fetch(url);
  const data = await res.json();

  return data.daily;
}

⚙️ 3. LÓGICA DA SIMULAÇÃO (COM DIAS + SATURAÇÃO)
🟢 Função principal
function simularChuvaAvancada(mm, dias, saturado) {
  const fatorSaturacao = saturado ? 1.5 : 1;

  grid.forEach(c => {
    const coef = getCoeficiente(c);

    // chuva acumulada ao longo dos dias
    const chuvaTotal = mm * dias;

    // redução de infiltração se solo saturado
    const infiltracao = saturado ? 0.1 : 0.4;

    c.agua = chuvaTotal * coef * fatorSaturacao * (1 - infiltracao);
    c.acumulado = 0;
  });

  propagarAgua();
  renderMapa();
}
🟢 Botões
btnSimular.onclick = () => {
  const mm = Number(chuvaManual.value);
  const dias = Number(diasInput.value);
  const saturado = soloSaturado.checked;

  simularChuvaAvancada(mm, dias, saturado);
};

btnPrevisao.onclick = async () => {
  const previsao = await getPrevisao(-20.51, -43.69);

  const chuvaTotal = previsao.precipitation_sum.reduce((a, b) => a + b, 0);

  simularChuvaAvancada(chuvaTotal, previsao.time.length, true);
};
💧 5. ANIMAÇÃO DA ÁGUA
function animarAgua() {
  grid.forEach(c => {
    if (!c.fluxo) return;

    const linha = L.polyline([c.center, c.fluxo.center], {
      color: "cyan",
      weight: 2,
      opacity: 0.7
    }).addTo(map);

    let offset = 0;

    setInterval(() => {
      offset += 1;
      linha.setStyle({
        dashArray: "5,10",
        dashOffset: offset
      });
    }, 100);
  });
}

btnAnimar.onclick = animarAgua;

📄 7. EXPORTAR RELATÓRIO TÉCNICO
function gerarRelatorio() {
  const pontosCriticos = grid.filter(c => c.acumulado > 200);

  const texto = `
RELATÓRIO DE SIMULAÇÃO DE CHUVA

Precipitação: ${chuvaManual.value} mm
Dias: ${diasInput.value}
Solo saturado: ${soloSaturado.checked ? "Sim" : "Não"}

Pontos críticos: ${pontosCriticos.length}
`;

  const blob = new Blob([texto], { type: "text/plain" });
  const link = document.createElement("a");

  link.href = URL.createObjectURL(blob);
  link.download = "relatorio.txt";
  link.click();
}

btnRelatorio.onclick = gerarRelatorio;
