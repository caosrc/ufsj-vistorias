import L from "leaflet";

export function criarMapa() {

  const map = L.map("map").setView([-20.51, -43.69], 12);

  L.tileLayer(
    "https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png"
  ).addTo(map);

  return map;
}
🧱 tiles.js (CIDADE INTEIRA SEM TRAVAR)
export function gerarTiles(bounds, size = 0.01) {

  const tiles = [];

  for (let lat = bounds.getSouth(); lat < bounds.getNorth(); lat += size) {
    for (let lng = bounds.getWest(); lng < bounds.getEast(); lng += size) {

      tiles.push({
        south: lat,
        north: lat + size,
        west: lng,
        east: lng + size
      });
    }
  }

  return tiles;
}
🌍 dem.js (CORRETO)
import { fromUrl } from "geotiff";

export async function carregarDEM(url) {

  const tiff = await fromUrl(url);
  const img = await tiff.getImage();

  const raster = await img.readRasters();

  return {
    data: raster[0],
    width: img.getWidth(),
    height: img.getHeight(),
    bbox: img.getBoundingBox()
  };
}
🌧️ chuva.js (REAL + HISTÓRICO SIMPLES)
export async function chuvaReal(lat, lon) {

  const res = await fetch(
    `https://api.open-meteo.com/v1/forecast?latitude=${lat}&longitude=${lon}&hourly=precipitation`
  );

  const data = await res.json();

  return data.hourly.precipitation[0] || 10;
}

// simula solo saturado
export function chuvaAcumulada(base) {

  return base * 1.5; // aumenta intensidade
}
🕳️ drenagem.js (REALISTA)
export function aplicarDrenagem(grid) {

  grid.forEach(c => {

    if (c.tipo === "rua") {

      const capacidade = 40;

      c.agua = Math.max(0, c.agua - capacidade);
    }

  });
}
⚡ WORKER (🔥 MAIS IMPORTANTE)
public/workers/simWorker.js
self.onmessage = function(e) {

  const { grid, chuva } = e.data;

  for (let i = 0; i < 5; i++) {

    grid.forEach(c => {

      if (!c.fluxo) return;

      const slope = c.elev - c.fluxo.elev;

      const vel = Math.sqrt(Math.max(slope, 0.001)) * 3;

      const fluxo = c.agua * (vel / 10);

      c.fluxo.agua += fluxo;
      c.agua -= fluxo;

    });

  }

  self.postMessage(grid);
};

import L from "leaflet";

export function render(grid, map) {

  grid.forEach(c => {

    if (c.agua > 50) {

      L.circle([c.lat, c.lng], {
        radius: 10,
        color: "blue",
        fillOpacity: 0.5
      }).addTo(map);

    }

  });
}

export function iniciar3D(container) {

  const scene = new THREE.Scene();

  const camera = new THREE.PerspectiveCamera(75, window.innerWidth/window.innerHeight, 0.1, 1000);

  const renderer = new THREE.WebGLRenderer();
  renderer.setSize(window.innerWidth, window.innerHeight);

  container.appendChild(renderer.domElement);

  camera.position.z = 100;

  function animate() {
    requestAnimationFrame(animate);
    renderer.render(scene, camera);
  }

  animate();

  return { scene };
}
🧠 main.js (TUDO JUNTO E OTIMIZADO)
import { criarMapa } from "./map.js";
import { gerarTiles } from "./tiles.js";
import { carregarDEM } from "./dem.js";
import { chuvaReal, chuvaAcumulada } from "./chuva.js";
import { aplicarDrenagem } from "./drenagem.js";
import { render } from "./renderer2D.js";

const map = criarMapa();

document.getElementById("simular").onclick = async () => {

  const bounds = map.getBounds();

  const tiles = gerarTiles(bounds);

  const chuvaBase = await chuvaReal(-20.51, -43.69);
  const chuva = chuvaAcumulada(chuvaBase);

  for (const tile of tiles) {

    const grid = [];

    const res = 0.0003;

    for (let lat = tile.south; lat < tile.north; lat += res) {
      for (let lng = tile.west; lng < tile.east; lng += res) {

        grid.push({
          lat,
          lng,
          elev: Math.random() * 100, // substitui pelo DEM
          agua: chuva,
          tipo: "rua"
        });

      }
    }

    // fluxo simples
    grid.forEach(c => {
      c.fluxo = grid[Math.floor(Math.random() * grid.length)];
    });

    const worker = new Worker("/workers/simWorker.js");

    worker.postMessage({ grid, chuva });

    worker.onmessage = (e) => {

      const resultado = e.data;

      aplicarDrenagem(resultado);

      render(resultado, map);

    };

  }

};
